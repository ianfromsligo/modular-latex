\begin{theorem}[1D non-existence within the search bound]
\label{thm:1d-nonexistence}
An exhaustive search over $\rho \in (0,1)$, $\delta \in [0, 2\pi)$,
and revival step count $N \le \Nmax$ finds no non-trivial full-state
revival for any $k \notin \{2, 3, 4, 5, 6, 8, 10\}$ with $k \le \Kmax$.
The completeness of the 1D classification for $k \ge 11$ remains open.
\end{theorem}
